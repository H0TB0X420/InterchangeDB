# Status tracking through inserts, deletes, and overwrites.

# Empty tree.
status
---
keys: 0
size: 0
tombstones: 0

# After inserting two keys: "aa"(2) + "11"(2) + "bb"(2) + "22"(2) = 8 bytes.
put aa 11
put bb 22
status
---
keys: 2
size: 8
tombstones: 0

# Delete one key: 1 key left, size drops by 4.
delete aa
status
---
keys: 1
size: 4
tombstones: 0

# Overwrite: key count unchanged, size adjusts for new value length.
# "bb"(2) + "longer"(6) = 8 bytes.
put bb longer
status
---
keys: 1
size: 8
tombstones: 0
