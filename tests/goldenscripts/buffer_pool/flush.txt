# Flush: write dirty pages to disk explicitly.

# Allocate and write.
new_page
write 0 first
---
page_id: 0

# Flush specific page.
flush 0
stats
---

# Write new data, then flush all.
write 0 second
flush

---

# Read back to confirm.
read 0
---
